# WAMR (wasm-micro-runtime, tools/setup-ios.sh) as a static library: the fast interpreter with SIMD (through
# SIMDe) and nothing else: no AOT, no JIT, no libc or WASI of its own (every import is the page's JS), software
# bounds checks (no signal handlers beside V8's). js/wasm_interp.cpp builds WebAssembly on it where V8 has none.
set(WAMR_ROOT_DIR ${TP}/wamr)
if(WIN32)
  set(WAMR_BUILD_PLATFORM windows)
elseif(APPLE)
  set(WAMR_BUILD_PLATFORM darwin)
else()
  set(WAMR_BUILD_PLATFORM linux)
endif()
if(CMAKE_OSX_ARCHITECTURES MATCHES "x86_64" OR (NOT CMAKE_OSX_ARCHITECTURES AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64"))
  set(WAMR_BUILD_TARGET X86_64)
else()
  set(WAMR_BUILD_TARGET AARCH64)
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_SIMD 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_BUILD_MINI_LOADER 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_STACK_HW_BOUND_CHECK 1)
set(FETCHCONTENT_SOURCE_DIR_SIMDE ${TP}/simde CACHE PATH "" FORCE)  # pinned, never fetched
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(wamr STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(wamr PUBLIC ${WAMR_ROOT_DIR}/core/iwasm/include)
target_compile_definitions(wamr INTERFACE HOST_HAS_WAMR)
